function capture(file: string, minutes: number[]) {
  for (const minute of minutes)
    db.insert(rawMessages)
      .values({
        sourceFile: file,
        uuid: `${file}-${minute}`,
        type: 'user',
        timestamp: new Date(Date.UTC(2026, 2, 4, 10, minute)).toISOString(),
        claudeSessionId: file,
        cwd: 'C:\\fixture'
      })
      .run()
}
function invoice(clientId: number, sessionId: number) {
  return invoiceService.saveInvoice({
    clientId,
    stripeInvoiceId: 'in_fixture',
    status: 'open',
    amountDueCents: 10000,
    amountPaidCents: 0,
    currency: 'usd',

        amountCents: 10000,

        sortOrder: 0
      }

  })
}

it('review repro: prompt timeline follows the explicitly selected activity', async () => {
  const sourceFile = '/review/map-timeline';
  capture(sourceFile, [0, 10, 40, 45, 50]);
  const rows = [0, 40].map(minute => db.insert(sessions).values({
    sourceFile, claudeSessionId: sourceFile, projectPath: 'C:\\fixture',
    startedAt: new Date(Date.UTC(2026, 2, 4, 10, minute)).toISOString(),
    endedAt: new Date(Date.UTC(2026, 2, 4, 10, minute + 10)).toISOString(),
    durationMinutes: 10, promptCount: 100
  }).returning().get());
  await sessionService.rebuildSessionsFromRaw();
  const review = sessionService.getReconciliationCases()[0];
  sessionService.mapSavedHistory(sourceFile, review.fingerprint!, [
    {sessionId: rows[0].id, detectedIndex: 1},
    {sessionId: rows[1].id, detectedIndex: 0}
  ]);
  expect(sessionService.getSessionById(rows[0].id)?.promptCount).toBe(3);
  const timeline = await sessionService.getPromptTimings(rows[0].id);
  expect(timeline).toHaveLength(3);
  expect(timeline[0].timestamp).toBe('2026-03-04T10:40:00.000Z');
});
