function capture(file: string, minutes: number[]) {
  for (const minute of minutes)
    db.insert(rawMessages)
      .values({
        sourceFile: file,
        uuid: `${file}-${minute}`,
        type: 'user',
        timestamp: new Date(Date.UTC(2026, 2, 4, 10, minute)).toISOString(),
        claudeSessionId: file,
        cwd: 'C:\\fixture'
      })
      .run()
}

it('review: adopting one source does not implicitly approve another pending source for the same legacy history', async () => {
  const first = '/review/first-copy'
  const second = '/review/second-copy'
  const row = db.insert(sessions).values({
    sourceFile: null, claudeSessionId: 'shared-conversation', projectPath: 'C:\\fixture',
    startedAt: '2026-03-04T10:00:00.000Z', endedAt: '2026-03-04T10:10:00.000Z', durationMinutes: 10
  }).returning().get()
  capture(first, [0, 10])
  capture(second, [0, 10])
  db.update(rawMessages).set({ claudeSessionId: 'shared-conversation' }).run()
  expect((await sessionService.rebuildSessionsFromRaw()).errors).toHaveLength(2)
  expect(sessionService.getAllSessions()).toHaveLength(1)
  const review = sessionService.getReconciliationCases().find((item) => item.sourceFile === first)!
  sessionService.mapSavedHistory(first, review.fingerprint!, [{ sessionId: row.id, detectedIndex: 0 }])
  expect(sessionService.getReconciliationCases()).toHaveLength(1)
  await sessionService.rebuildSessionsFromRaw()
  expect(sessionService.getAllSessions()).toHaveLength(1)
  expect(sessionService.getReconciliationCases()).toEqual([expect.objectContaining({ sourceFile: second })])
})
it('review: blocks invoice previews after a source-less candidate is reassigned', async () => {
  const sourceFile = '/review/invoice-source-less'
  const firstClient = clientProjectService.createClient({ name: 'First', billableRate: 60 })
  const nextClient = clientProjectService.createClient({ name: 'Reassigned', billableRate: 60 })
  const row = db.insert(sessions).values({
    sourceFile: null, claudeSessionId: sourceFile, projectPath: 'C:\\fixture', clientId: firstClient.id,
    startedAt: '2026-03-04T10:00:00.000Z', endedAt: '2026-03-04T10:10:00.000Z', durationMinutes: 10
  }).returning().get()
  capture(sourceFile, [0, 10])
  expect((await sessionService.rebuildSessionsFromRaw()).errors).toHaveLength(1)
  sessionService.updateSession(row.id, { clientId: nextClient.id })
  expect(sessionService.getReconciliationCases()).toHaveLength(1)
  await expect(invoiceService.generateLineItems(nextClient.id, '2026-03-01', '2026-03-10')).rejects.toThrow('Review unresolved history')
})
